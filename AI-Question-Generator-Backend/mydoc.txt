npm init-y
#initialize as node.js project. Create package.js


npm install express cors body-parser dotenv
# express → The web framework (lets us build an API server).
cors → Middleware to allow the frontend (React) to communicate with the backend from another port.
body-parser → Helps parse JSON input in requests (req.body).
dotenv → Lets to use environment variables (like API keys) securely via a .env file


npm install --save-dev nodemon
#auto restart on file change, install nodemon
add to package.json:

"scripts": {
  "start": "node server.js",
  "dev": "nodemon server.js"
}
